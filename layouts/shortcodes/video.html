{{- /* Embeds a video file from the page bundle with the native HTML5 player. {{< video src="demo.mp4" >}} {{< video src="demo.mp4" poster="cover.jpg" title="A walkthrough" >}} preload="none" keeps the file from downloading until the reader presses play, so large recordings don't weigh down the page. The poster is shown until then. Videos are muted by default; pass muted="false" for ones with sound. */ -}} {{- $src := .Get "src" -}} {{- $video := .Page.Resources.GetMatch $src -}} {{- if not $video -}} {{- errorf "video shortcode: can't find %q in page resources. See %s" $src .Position -}} {{- end -}} {{- $poster := "" -}} {{- with .Get "poster" -}} {{- with $.Page.Resources.GetMatch . -}} {{- $poster = .RelPermalink -}} {{- else -}} {{- errorf "video shortcode: can't find poster %q in page resources. See %s" . $.Position -}} {{- end -}} {{- end -}} {{- $muted := ne (.Get "muted") "false" -}} {{- with $video -}} {{- end -}}
