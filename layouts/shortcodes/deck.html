{{- /* Shows a slide deck and a video of it in one frame, with a switch so the reader can either click through the live slides or watch the recording. {{< deck slides="https://example.com/slides/1?embedded=true" video="walkthrough.mp4" >}} {{< deck slides="https://example.com/slides/1?embedded=true" open="https://example.com/slides/1" video="walkthrough.mp4" poster="poster.jpg" title="My deck" default="video" >}} slides URL of the live deck, shown in an iframe open URL behind "Open slides" (defaults to slides); set it when slides carries embed-only parameters video video file in the page bundle; poster is an optional image from it default which view shows first: "slides" (default) or "video" hint text shown next to the switch in the slides view The iframe only loads once the slides view is shown, so readers who only watch the video don't download the deck. Without JavaScript both views are shown one below the other. The script is loaded by layouts/partials/extend_head.html on pages that use this shortcode. */ -}} {{- $slides := .Get "slides" -}} {{- if not $slides -}} {{- errorf "deck shortcode: slides is required. See %s" .Position -}} {{- end -}} {{- $video := .Page.Resources.GetMatch (.Get "video") -}} {{- if not $video -}} {{- errorf "deck shortcode: can't find video %q in page resources. See %s" (.Get "video") .Position -}} {{- end -}} {{- $poster := "" -}} {{- with .Get "poster" -}} {{- with $.Page.Resources.GetMatch . -}} {{- $poster = .RelPermalink -}} {{- else -}} {{- errorf "deck shortcode: can't find poster %q in page resources. See %s" . $.Position -}} {{- end -}} {{- end -}} {{- $open := .Get "open" | default $slides -}} {{- $title := .Get "title" | default "Slides" -}} {{- $hint := .Get "hint" | default "click, then use ← → to step through" -}} {{- $view := .Get "default" | default "slides" -}} {{- if not (in (slice "slides" "video") $view) -}} {{- errorf "deck shortcode: default must be \"slides\" or \"video\", got %q. See %s" $view .Position -}} {{- end -}} {{- $id := printf "deck-%d" .Ordinal -}}
